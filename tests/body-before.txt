        break;
      case 2:

      case 3:

        Acebott.Move(Move_Right, 180);
        delay(500);

void runModule(int device) {
  val = readBuffer(12);
  switch (device) {
    case 0x0C:
      {
        switch (val) {
          case 0x01: Acebott.Move(Forward, speeds); break;
          case 0x02: Acebott.Move(Backward, speeds); break;
          case 0x03: Acebott.Move(Move_Left, speeds); break;
          case 0x04: Acebott.Move(Move_Right, speeds); break;
          case 0x05: Acebott.Move(Top_Left, speeds); break;
          case 0x06: Acebott.Move(Bottom_Left, speeds); break;
          case 0x07: Acebott.Move(Top_Right, speeds); break;
          case 0x08: Acebott.Move(Bottom_Right, speeds); break;
          case 0x0A: Acebott.Move(Clockwise, speeds); break;
          case 0x09: Acebott.Move(Contrarotate, speeds); break;
          case 0x00: Acebott.Move(Stop, 0); break;
          default: break;
        }
      }
      break;
    case 0x02: // Dispositivo 2: Servo de Cámara
    case 35:   // Dispositivo 35: Atendido también para control de servo
      {
        newRunMode = 0;
        Servo_Move(val);
      }
      break;
    case 0x03:
      Buzzer_run(val);
      break;
    case 0x05:
      digitalWrite(LED_Module1, val);
      digitalWrite(LED_Module2, val);
      break;
    case 0x08:
      digitalWrite(Shoot_PIN, HIGH);
      delay(200);
      digitalWrite(Shoot_PIN, LOW);
      break;
    case 0x0D:
      speeds = val;
      break;
  }
}

void parseData() {
  isStart = false;
  int action = readBuffer(9);
  int device = readBuffer(10);
  int index2 = readBuffer(2);

  Serial.print("Accion: ");
  Serial.print(action);
  Serial.print(" | Dispositivo: ");
  Serial.print(device);
  Serial.print(" | Valor: ");
  Serial.println(readBuffer(12));

  // Si se presiona un botón de modo automático (Acciones 4, 5, 6, 7 u 8)
  if (action >= 4 && action <= 8) {
    newRunMode = action;
  }

  switch (action) {
    case CMD_RUN:
      if (device != 2 && device != 35) { 
        function_mode = STANDBY;
      }
      runModule(device);
      break;
    case CMD_STANDBY:
      function_mode = STANDBY;
      newRunMode = 0;
      Acebott.Move(Stop, 0);
      turnServo.write(90);
      fixedServo.write(90);
      break;
    default: 
      break;
  }
}

void RXpack_func() {
  if (Serial.available() > 0) {
    unsigned char c = Serial.read() & 0xff;
    Serial.write(c);

    if (c == 0x55 && isStart == false) {
      if (prevc == 0xff) {
        index_a = 1;
        isStart = true;
      }
    } else {
      prevc = c;
      if (isStart) {
        if (index_a == 2) {
          dataLen = c;
        } else if (index_a > 2) {
          dataLen--;
        }
        writeBuffer(index_a, c);
      }
    }
    index_a++;
    if (index_a > 120) {
      index_a = 0;
      isStart = false;
    }
    if (isStart && dataLen == 0 && index_a > 3) {
      isStart = false;
      parseData();
      index_a = 0;
    }
  }
}
